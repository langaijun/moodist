### 1.1 主应用组件 (src/components/app/app.tsx)

```typescript
/**
 * Moodist 主应用组件
 *
 * 职责：
 * 1. 管理应用的整体结构和布局
 * 2. 协调各个子组件的交互
 * 3. 处理浏览器音频上下文的生命周期
 * 4. 响应系统级事件（如页面可见性变化）
 * 5. 管理音效分类和收藏的动态展示
 */

import { useMemo, useEffect } from 'react';
import { useShallow } from 'zustand/react/shallow';
import { BiSolidHeart } from 'react-icons/bi/index';
import { Howler } from 'howler';

import { useSoundStore } from '@/stores/sound';

// ... 其他导入

export function App() {
  // 获取所有音效分类（使用useMemo避免重复计算）
  const categories = useMemo(() => sounds.categories, []);

  // 使用浅比较选择器，只订阅需要的store部分
  const favorites = useSoundStore(useShallow(state => state.getFavorites()));
  const pause = useSoundStore(state => state.pause);
  const lock = useSoundStore(state => state.lock);
  const unlock = useSoundStore(state => state.unlock);

  /**
   * 动态生成收藏音效列表
   * 根据收藏ID从所有分类中筛选对应的音效对象
   * 按照收藏的顺序重新排列
   */
  const favoriteSounds = useMemo(() => {
    const favoriteSounds = categories
      .map(category => category.sounds)
      .flat()
      .filter(sound => favorites.includes(sound.id));

    // 按照收藏列表的顺序重新排序
    return favorites.map(favorite =>
      favoriteSounds.find(sound => sound.id === favorite),
    );
  }, [favorites, categories]);

  /**
   * 处理浏览器页面可见性变化
   * 当页面从隐藏变为可见时，恢复音频上下文
   * 这是为了解决某些浏览器在后台时暂停音频上下文的问题
   */
  useEffect(() => {
    const onChange = () => {
      const { ctx } = Howler;

      if (ctx && !document.hidden) {
        // 延迟100ms恢复，确保浏览器完全切换回来
        setTimeout(() => {
          ctx.resume();
        }, 100);
      }
    };

    document.addEventListener('visibilitychange', onChange, false);

    // 清理事件监听器
    return () => document.removeEventListener('visibilitychange', onChange);
  }, []);

  /**
   * 监听淡出事件
   * 当收到淡出事件时，锁定状态并延迟暂停播放
   * 用于睡眠定时器等功能
   */
  useEffect(() => {
    const unsubscribe = subscribe(FADE_OUT, (e: { duration: number }) => {
      lock();  // 锁定状态，防止用户操作

      setTimeout(() => {
        pause();  // 延迟暂停
        unlock(); // 解锁状态
      }, e.duration);
    });

    return unsubscribe;  // 清理订阅
  }, [pause, lock, unlock]);

  /**
   * 构建完整的分类列表
   * 如果有收藏音效，在列表开头添加收藏分类
   */
  const allCategories = useMemo(() => {
    const favorites = [];

    if (favoriteSounds.length) {
      favorites.push({
        icon: <BiSolidHeart />,
        id: 'favorites',
        sounds: favoriteSounds as Array<Sound>,
        title: 'Favorites',
      });
    }

    return [...favorites, ...categories];
  }, [favoriteSounds, categories]);

  return (
    <SnackbarProvider>
      <StoreConsumer>
        <MediaControls />        {/* 媒体会话控制 */}
        <Container>
          <div id="app" />
          <Buttons />            {/* 播放控制按钮 */}
          <Categories categories={allCategories} />  {/* 音效分类 */}
        </Container>

        <Toolbar />             {/* 底部工具栏 */}
        <SharedModal />         {/* 共享模态框 */}
      </StoreConsumer>
    </SnackbarProvider>
  );
}
```

#### 1.2.1 状态定义 (sound.state.ts)

```typescript
/**
 * 音效状态定义
 *
 * 定义了Zustand store的状态结构和方法签名
 * 所有音效相关的状态都集中管理
 */

import type { StateCreator } from 'zustand';
import type { SoundActions } from './sound.actions';
import { sounds } from '@/data/sounds';

export interface SoundState {
  /**
   * 获取收藏的音效ID列表
   * @returns {string[]} 收藏的音效ID数组
   */
  getFavorites: () => Array<string>;

  /**
   * 全局音量控制
   * 范围：0-1，默认值1
   */
  globalVolume: number;

  /**
   * 历史状态快照
   * 用于撤销操作，保存之前的完整状态
   */
  history: {
    [id: string]: {
      isFavorite: boolean;
      isSelected: boolean;
      volume: number;
    };
  } | null;

  /**
   * 播放状态
   * true: 正在播放
   * false: 暂停中
   */
  isPlaying: boolean;

  /**
   * 锁定状态
   * true: 锁定，禁止用户操作（用于动画过渡等场景）
   * false: 解锁，允许用户操作
   */
  locked: boolean;

  /**
   * 检查是否没有选中任何音效
   * @returns {boolean} true表示没有选中音效
   */
  noSelected: () => boolean;

  /**
   * 所有音效的状态映射
   * key: 音效ID
   * value: 音效的状态对象
   */
  sounds: {
    [id: string]: {
      isFavorite: boolean;   // 是否收藏
      isSelected: boolean;   // 是否选中
      volume: number;        // 音量 0-1
    };
  };
}

/**
 * 创建初始状态
 *
 * 根据音效数据文件初始化所有音效的状态
 * 每个音效默认状态：未选中、未收藏、音量0.5
 */
export const createState: StateCreator<
  SoundState & SoundActions,
  [],
  [],
  SoundState
> = (set, get) => {
  // 初始化状态对象
  const state: SoundState = {
    getFavorites() {
      const { sounds } = get();
      const ids = Object.keys(sounds);
      // 筛选出所有收藏的音效ID
      const favorites = ids.filter(id => sounds[id].isFavorite);

      return favorites;
    },
    globalVolume: 1,
    history: null,
    isPlaying: false,
    locked: false,
    noSelected() {
      const { sounds } = get();
      const keys = Object.keys(sounds);

      // 检查是否所有音效都未选中
      return keys.every(key => !sounds[key].isSelected);
    },
    sounds: {},
  };

  // 遍历所有分类和音效，初始化状态
  const { categories } = sounds;

  categories.forEach(category => {
    const { sounds } = category;

    sounds.forEach(sound => {
      state.sounds[sound.id] = {
        isFavorite: false,
        isSelected: false,
        volume: 0.5,
      };
    });
  });

  return state;
};
```

```typescript
/**
 * 音效操作方法定义
 *
 * 定义了所有修改音效状态的方法
 * 这些方法会被注入到Zustand store中
 */

import type { StateCreator } from 'zustand';
import type { SoundState } from './sound.state';
import { pickMany, random } from '@/helpers/random';

export interface SoundActions {
  /**
   * 锁定状态
   * 用于防止用户在特定操作期间进行其他操作
   */
  lock: () => void;

  /**
   * 覆盖当前状态
   * 用于从分享链接或其他方式恢复音效配置
   * @param {Record<string, number>} sounds - 音效ID到音量的映射
   */
  override: (sounds: Record<string, number>) => void;

  /**
   * 暂停播放
   */
  pause: () => void;

  /**
   * 开始播放
   */
  play: () => void;

  /**
   * 恢复历史状态
   * 用于撤销操作
   */
  restoreHistory: () => void;

  /**
   * 选择音效
   * @param {string} id - 音效ID
   */
  select: (id: string) => void;

  /**
   * 设置全局音量
   * @param {number} volume - 音量值 0-1
   */
  setGlobalVolume: (volume: number) => void;

  /**
   * 设置单个音效音量
   * @param {string} id - 音效ID
   * @param {number} volume - 音量值 0-1
   */
  setVolume: (id: string, volume: number) => void;

  /**
   * 随机选择音效
   * 随机选择4个音效，设置随机音量并开始播放
   */
  shuffle: () => void;

  /**
   * 切换收藏状态
   * @param {string} id - 音效ID
   */
  toggleFavorite: (id: string) => void;

  /**
   * 切换播放状态
   */
  togglePlay: () => void;

  /**
   * 解锁状态
   */
  unlock: () => void;

  /**
   * 取消选择音效
   * @param {string} id - 音效ID
   */
  unselect: (id: string) => void;

  /**
   * 取消所有选择
   * @param {boolean} pushToHistory - 是否保存到历史记录
   */
  unselectAll: (pushToHistory?: boolean) => void;
}

/**
 * 创建操作方法
 * 这些方法会访问和修改store状态
 */
export const createActions: StateCreator<
  SoundActions & SoundState,
  [],
  [],
  SoundActions
> = (set, get) => {
  return {
    /**
     * 锁定状态
     */
    lock() {
      set({ locked: true });
    },

    /**
     * 覆盖当前状态
     * 用于从分享链接或其他方式恢复音效配置
     */
    override(newSounds) {
      get().unselectAll();  // 先清除当前选择

      const sounds = get().sounds;

      // 根据传入的配置设置音效状态
      Object.keys(newSounds).forEach(sound => {
        if (sounds[sound]) {
          sounds[sound].isSelected = true;
          sounds[sound].volume = newSounds[sound];
        }
      });

      set({ history: null, sounds: { ...sounds } });
    },

    /**
     * 暂停播放
     */
    pause() {
      set({ isPlaying: false });
    },

    /**
     * 开始播放
     */
    play() {
      set({ isPlaying: true });
    },

    /**
     * 恢复历史状态
     */
    restoreHistory() {
      const history = get().history;

      if (!history) return;

      set({ history: null, sounds: history });
    },

    /**
     * 选择音效
     */
    select(id) {
      set({
        history: null,  // 清除历史记录
        sounds: {
          ...get().sounds,
          [id]: { ...get().sounds[id], isSelected: true },
        },
      });
    },

    /**
     * 设置全局音量
     */
    setGlobalVolume(volume) {
      set({
        globalVolume: volume,
      });
    },

    /**
     * 设置单个音效音量
     */
    setVolume(id, volume) {
      set({
        sounds: {
          ...get().sounds,
          [id]: { ...get().sounds[id], volume },
        },
      });
    },

    /**
     * 随机选择音效
     * 随机选择4个音效，设置随机音量并开始播放
     */
    shuffle() {
      const sounds = get().sounds;
      const ids = Object.keys(sounds);

      // 重置所有音效状态
      ids.forEach(id => {
        sounds[id].isSelected = false;
        sounds[id].volume = 0.5;
      });

      // 随机选择4个音效
      const randomIDs = pickMany(ids, 4);

      randomIDs.forEach(id => {
        sounds[id].isSelected = true;
        sounds[id].volume = random(0.2, 1);  // 随机音量 0.2-1
      });

      set({ history: null, isPlaying: true, sounds });
    },

    /**
     * 切换收藏状态
     */
    toggleFavorite(id) {
      const sounds = get().sounds;
      const sound = sounds[id];

      set({
        history: null,
        sounds: {
          ...sounds,
          [id]: { ...sound, isFavorite: !sound.isFavorite },
        },
      });
    },

    /**
     * 切换播放状态
     */
    togglePlay() {
      set({ isPlaying: !get().isPlaying });
    },

    /**
     * 解锁状态
     */
    unlock() {
      set({ locked: false });
    },

    /**
     * 取消选择音效
     */
    unselect(id) {
      set({
        sounds: {
          ...get().sounds,
          [id]: { ...get().sounds[id], isSelected: false },
        },
      });
    },

    /**
     * 取消所有选择
     */
    unselectAll(pushToHistory = false) {
      const noSelected = get().noSelected();

      if (noSelected) return;  // 如果已经没有选中，直接返回

      const sounds = get().sounds;

      // 如果需要保存到历史记录
      if (pushToHistory) {
        const history = JSON.parse(JSON.stringify(sounds));
        set({ history });
      }

      // 重置所有音效状态
      const ids = Object.keys(sounds);

      ids.forEach(id => {
        sounds[id].isSelected = false;
        sounds[id].volume = 0.5;
      });

      set({ sounds });
    },
  };
};
```

```typescript
/**
 * 音效状态Store配置
 *
 * 配置Zustand store的持久化、版本控制等
 */

import { create } from 'zustand';
import { createJSONStorage, persist } from 'zustand/middleware';
import merge from 'deepmerge';

import { type SoundState, createState } from './sound.state';
import { type SoundActions, createActions } from './sound.actions';

/**
 * 创建音效状态Store
 *
 * 使用persist中间件实现状态持久化
 * 支持版本控制和数据迁移
 */
export const useSoundStore = create<SoundState & SoundActions>()(
  persist(
    (...a) => ({
      ...createState(...a),    // 合并初始状态
      ...createActions(...a),  // 合并操作方法
    }),
    {
      /**
       * 合并持久化数据和当前状态
       * 使用deepmerge确保深层对象正确合并
       */
      merge: (persisted, current) =>
        merge(
          current,              // 当前状态优先
          persisted as any,     // 持久化状态其次
        ),

      /**
       * 存储配置
       */
      name: 'moodist-sounds',  // 存储键名
      partialize: state => ({  // 只持久化部分状态
        globalVolume: state.globalVolume,
        sounds: state.sounds,
      }),
      skipHydration: true,     // 跳过自动水合，手动控制
      storage: createJSONStorage(() => localStorage),  // 使用localStorage
      version: 0,              // 版本号
    },
  ),
);
```

### 1.3 音效播放Hook (src/hooks/use-sound.ts)

```typescript
/**
 * 音效播放Hook
 *
 * 封装Howler.js的使用，提供统一的音效播放控制
 * 支持播放、暂停、停止、音量控制、淡出等功能
 */

import { useMemo, useEffect, useCallback, useState } from 'react';
import { Howl } from 'howler';

import { useLoadingStore } from '@/stores/loading';
import { subscribe } from '@/lib/event';
import { useSSR } from './use-ssr';
import { FADE_OUT } from '@/constants/events';

/**
 * useSound Hook
 *
 * @param {string} src - 音效文件URL
 * @param {Object} options - 播放选项
 * @param {boolean} options.loop - 是否循环播放
 * @param {boolean} options.preload - 是否预加载
 * @param {number} options.volume - 初始音量
 * @param {boolean} html5 - 是否使用HTML5 Audio
 * @returns {UseSoundReturn} 音效控制对象
 */
export function useSound(
  src: string,
  options: { loop?: boolean; preload?: boolean; volume?: number } = {},
  html5: boolean = false,
) {
  // 加载状态
  const [hasLoaded, setHasLoaded] = useState(false);
  const isLoading = useLoadingStore(state => state.loaders[src]);
  const setIsLoading = useLoadingStore(state => state.set);

  const { isBrowser } = useSSR();

  /**
   * 创建Howl实例
   * 只在浏览器环境中创建，避免SSR问题
   */
  const sound = useMemo<Howl | null>(() => {
    let sound: Howl | null = null;

    if (isBrowser) {
      sound = new Howl({
        html5,  // 使用HTML5 Audio或Web Audio API
        onload: () => {  // 加载完成回调
          setIsLoading(src, false);
          setHasLoaded(true);
        },
        preload: options.preload ?? false,  // 是否预加载
        src: src,
      });

      // 设置音频会话类型（iOS支持）
      if (window.navigator.audioSession) {
        window.navigator.audioSession.type = 'playback';
      }
    }

    return sound;
  }, [src, isBrowser, setIsLoading, html5, options.preload]);

  /**
   * 响应循环播放选项变化
   */
  useEffect(() => {
    if (sound) {
      sound.loop(options.loop ?? false);
    }
  }, [sound, options.loop]);

  /**
   * 响应音量选项变化
   */
  useEffect(() => {
    if (sound) sound.volume(options.volume ?? 0.5);
  }, [sound, options.volume]);

  /**
   * 播放音效
   * @param {Function} cb - 播放完成回调
   */
  const play = useCallback(
    (cb?: () => void) => {
      if (sound) {
        // 如果音效还未加载且不在加载中，开始加载
        if (!hasLoaded && !isLoading) {
          setIsLoading(src, true);
          sound.load();
        }

        // 如果音效未在播放，开始播放
        if (!sound.playing()) {
          sound.play();
        }

        // 注册播放完成回调
        if (typeof cb === 'function') sound.once('end', cb);
      }
    },
    [src, setIsLoading, sound, hasLoaded, isLoading],
  );

  /**
   * 停止音效
   */
  const stop = useCallback(() => {
    if (sound) sound.stop();
  }, [sound]);

  /**
   * 暂停音效
   */
  const pause = useCallback(() => {
    if (sound) sound.pause();
  }, [sound]);

  /**
   * 淡出音效
   * @param {number} duration - 淡出时长（毫秒）
   */
  const fadeOut = useCallback(
    (duration: number) => {
      sound?.fade(sound.volume(), 0, duration);

      setTimeout(() => {
        pause();  // 淡出完成后暂停
        sound?.volume(options.volume || 0.5);  // 恢复音量
      }, duration);
    },
    [options.volume, sound, pause],
  );

  /**
   * 监听淡出事件
   * 当收到淡出事件时，执行淡出操作
   */
  useEffect(() => {
    const listener = (e: { duration: number }) => fadeOut(e.duration);

    return subscribe(FADE_OUT, listener);
  }, [fadeOut]);

  /**
   * 返回音效控制对象
   */
  const control = useMemo(
    () => ({ fadeOut, isLoading, pause, play, stop }),
    [play, stop, pause, isLoading, fadeOut],
  );

  return control;
}
```

### 1.4 音效组件 (src/components/sounds/sound/sound.tsx)

```typescript
/**
 * 单个音效组件
 *
 * 负责显示和交互单个音效的控制界面
 * 包含音效图标、名称、音量控制和收藏按钮
 */

import { useCallback, useEffect, forwardRef, useMemo } from 'react';
import { ImSpinner9 } from 'react-icons/im/index';

import { Range } from './range';
import { Favorite } from './favorite';

import { useSound } from '@/hooks/use-sound';
import { useSoundStore } from '@/stores/sound';
import { useLoadingStore } from '@/stores/loading';
import { cn } from '@/helpers/styles';

import styles from './sound.module.css';
import type { Sound as SoundType } from '@/data/types';
import { useKeyboardButton } from '@/hooks/use-keyboard-button';

/**
 * Sound组件接口
 */
interface SoundProps extends SoundType {
  functional: boolean;        // 是否启用功能交互
  hidden: boolean;            // 是否隐藏（用于"显示更多"功能）
  selectHidden: (key: string) => void;
  unselectHidden: (key: string) => void;
}

/**
 * 单个音效组件
 * 使用forwardRef支持ref转发，用于焦点管理
 */
export const Sound = forwardRef<HTMLDivElement, SoundProps>(function Sound(
  { functional, hidden, icon, id, label, selectHidden, src, unselectHidden },
  ref,
) {
  // 从store获取相关状态
  const isPlaying = useSoundStore(state => state.isPlaying);
  const play = useSoundStore(state => state.play);
  const selectSound = useSoundStore(state => state.select);
  const unselectSound = useSoundStore(state => state.unselect);
  const setVolume = useSoundStore(state => state.setVolume);
  const isSelected = useSoundStore(state => state.sounds[id].isSelected);
  const locked = useSoundStore(state => state.locked);

  const volume = useSoundStore(state => state.sounds[id].volume);
  const globalVolume = useSoundStore(state => state.globalVolume);

  /**
   * 计算实际音量
   * 实际音量 = 单个音效音量 × 全局音量
   */
  const adjustedVolume = useMemo(
    () => volume * globalVolume,
    [volume, globalVolume],
  );

  // 获取加载状态
  const isLoading = useLoadingStore(state => state.loaders[src]);

  // 使用音效播放Hook
  const sound = useSound(src, { loop: true, volume: adjustedVolume });

  /**
   * 响应播放状态变化
   * 当音效被选中且正在播放时，播放音效
   */
  useEffect(() => {
    if (locked) return;  // 锁定状态下不响应

    if (isSelected && isPlaying && functional) {
      sound?.play();
    } else {
      sound?.pause();
    }
  }, [isSelected, sound, isPlaying, functional, locked]);

  /**
   * 管理隐藏状态的通知
   * 当音效被隐藏但被选中时，通知父组件
   */
  useEffect(() => {
    if (hidden && isSelected) selectHidden(label);
    else if (hidden && !isSelected) unselectHidden(label);
  }, [label, isSelected, hidden, selectHidden, unselectHidden]);

  /**
   * 选择音效
   */
  const select = useCallback(() => {
    if (locked) return;  // 锁定状态下不允许操作
    selectSound(id);
    play();  // 自动开始播放
  }, [selectSound, play, id, locked]);

  /**
   * 取消选择音效
   */
  const unselect = useCallback(() => {
    if (locked) return;  // 锁定状态下不允许操作
    unselectSound(id);
    setVolume(id, 0.5);  // 重置音量为默认值
  }, [unselectSound, setVolume, id, locked]);

  /**
   * 切换选择状态
   */
  const toggle = useCallback(() => {
    if (locked) return;  // 锁定状态下不允许操作
    if (isSelected) unselect();
    else select();
  }, [isSelected, select, unselect, locked]);

  /**
   * 点击处理
   */
  const handleClick = useCallback(() => {
    toggle();
  }, [toggle]);

  /**
   * 键盘事件处理
   * 支持Enter和Space键
   */
  const handleKeyDown = useKeyboardButton(() => {
    toggle();
  });

  return (
    <div
      aria-label={`${label} sound`}
      ref={ref}
      role="button"
      tabIndex={0}  // 可聚焦
      className={cn(
        styles.sound,
        isSelected && styles.selected,  // 选中状态样式
        hidden && styles.hidden,          // 隐藏状态样式
      )}
      onClick={handleClick}
      onKeyDown={handleKeyDown}
    >
      <Favorite id={id} label={label} />

      <div className={styles.icon}>
        {isLoading ? (
          // 加载中显示spinner
          <span aria-hidden="true" className={styles.spinner}>
            <ImSpinner9 />
          </span>
        ) : (
          // 正常显示图标
          <span aria-hidden="true">{icon}</span>
        )}
      </div>

      <div className={styles.label} id={id}>
        {label}
      </div>

      <Range id={id} label={label} />
    </div>
  );
});
```

```typescript
/**
 * 通用模态框组件
 *
 * 提供模态框的通用功能：
 * - 背景遮罩
 * - 焦点陷阱
 * - 键盘关闭
 * - 动画效果
 * - 响应式设计
 */

import { useEffect } from 'react';
import { AnimatePresence, motion } from 'motion/react';
import { IoClose } from 'react-icons/io5/index';
import FocusTrap from 'focus-trap-react';

import { Portal } from '@/components/portal';
import { fade, mix, slideY } from '@/lib/motion';
import { cn } from '@/helpers/styles';

import styles from './modal.module.css';

/**
 * Modal组件接口
 */
interface ModalProps {
  children: React.ReactNode;
  lockBody?: boolean;    // 是否锁定背景滚动
  onClose: () => void;   // 关闭回调
  persist?: boolean;     // 是否持久化显示（不使用动画）
  show: boolean;         // 是否显示
  wide?: boolean;        // 是否宽模式
}

const TRANSITION_DURATION = 300;  // 动画时长（毫秒）

/**
 * Modal组件
 */
export function Modal({
  children,
  lockBody = true,
  onClose,
  persist = false,
  show,
  wide,
}: ModalProps) {
  // 动画变体配置
  const variants = {
    modal: mix(fade(), slideY(20)),  // 模态框：淡入 + 上滑
    overlay: fade(),                // 遮罩：淡入
  };

  /**
   * 处理背景滚动锁定
   * 模态框显示时锁定背景滚动，关闭时恢复
   */
  useEffect(() => {
    if (show && lockBody) {
      document.body.style.overflowY = 'hidden';
    } else if (lockBody) {
      // 等待动画完成后再恢复滚动
      setTimeout(() => {
        document.body.style.overflowY = 'auto';
      }, TRANSITION_DURATION);
    }
  }, [show, lockBody]);

  /**
   * 键盘事件处理
   * ESC键关闭模态框
   */
  useEffect(() => {
    function keyListener(e: KeyboardEvent) {
      if (show && e.key === 'Escape') {
        onClose();
      }
    }

    document.addEventListener('keydown', keyListener);

    return () => document.removeEventListener('keydown', keyListener);
  }, [onClose, show]);

  /**
   * 根据persist属性决定动画配置
   * persist模式：手动控制显示/隐藏
   * 非persist模式：使用AnimatePresence自动处理
   */
  const animationProps = persist
    ? {
        animate: show ? 'show' : 'hidden',
      }
    : {
        animate: 'show',
        exit: 'hidden',
        initial: 'hidden',
      };

  /**
   * 模态框内容
   */
  const content = (
    <FocusTrap active={show}>  {/* 焦点陷阱：确保焦点在模态框内 */}
      <div>
        <motion.div
          {...animationProps}
          className={styles.overlay}
          transition={{ duration: TRANSITION_DURATION / 1000 }}
          variants={variants.overlay}
          onClick={onClose}
          onKeyDown={onClose}
        />
        <div className={styles.modal}>
          <motion.div
            {...animationProps}
            className={cn(styles.content, wide && styles.wide)}
            transition={{ duration: TRANSITION_DURATION / 1000 }}
            variants={variants.modal}
          >
            <button className={styles.close} onClick={onClose}>
              <IoClose />
            </button>
            {children}
          </motion.div>
        </div>
      </div>
    </FocusTrap>
  );

  return (
    <Portal>
      {persist ? (
        // 持久化模式：手动控制显示
        <div style={{ display: show ? 'block' : 'none' }}>{content}</div>
      ) : (
        // 非持久化模式：使用AnimatePresence
        <AnimatePresence>{show && content}</AnimatePresence>
      )}
    </Portal>
  );
}
```

```typescript
 * 提供随机选择和随机数生成功能

/**
 * 从数组中随机选择多个元素
 * @param {Array<T>} array - 源数组
 * @param {number} count - 要选择的数量
 * @returns {Array<T>} 随机选择的元素数组
 */
export function pickMany<T>(array: Array<T>, count: number): Array<T> {
  // 洗牌算法
  const shuffled = [...array].sort(() => Math.random() - 0.5);
  return shuffled.slice(0, count);
}

/**
 * 生成指定范围内的随机数
 * @param {number} min - 最小值
 * @param {number} max - 最大值
 * @returns {number} 随机数
 */
export function random(min: number, max: number): number {
  return Math.random() * (max - min) + min;
}
```

### 3.2 事件系统 (src/lib/event.ts)

```typescript
/**
 * 简单的发布-订阅事件系统
 *
 * 提供组件间通信的解耦方式
 */

type EventListener = (data: unknown) => void;
type EventMap = Record<string, EventListener[]>;

/**
 * 事件监听器映射
 * key: 事件名称
 * value: 监听器数组
 */
const events: EventMap = {};

/**
 * 订阅事件
 * @param {string} event - 事件名称
 * @param {EventListener} listener - 事件监听器
 * @returns {Function} 取消订阅函数
 */
export function subscribe(event: string, listener: EventListener): () => void {
  if (!events[event]) {
    events[event] = [];
  }

  events[event].push(listener);

  // 返回取消订阅函数
  return () => unsubscribe(event, listener);
}

/**
 * 发布事件
 * @param {string} event - 事件名称
 * @param {unknown} data - 事件数据
 */
export function dispatch(event: string, data?: unknown): void {
  if (!events[event]) return;

  events[event].forEach(listener => {
    listener(data);
  });
}

/**
 * 取消订阅事件
 * @param {string} event - 事件名称
 * @param {EventListener} listener - 事件监听器
 */
export function unsubscribe(event: string, listener: EventListener): void {
  if (!events[event]) return;

  events[event] = events[event].filter(l => l !== listener);
}
```
